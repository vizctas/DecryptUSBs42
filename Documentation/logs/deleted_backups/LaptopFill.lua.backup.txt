-- Function to get translated message with fallback
local function getTranslatedMessage(key, fallback)
    if not key then return fallback or "Unknown message" end
    local translated = getText(key)
    if translated and translated ~= key then
        return translated
    end
    return fallback or "Unknown message"
end

local function ensureDecryptSkillDrive()
    if DecryptSkillDrive and type(DecryptSkillDrive.new) == "function" then
        return DecryptSkillDrive
    end

    local moduleNames = {
        "client/TimedActions/DecryptSkillDrive",
        "TimedActions/DecryptSkillDrive"
    }

    for _, name in ipairs(moduleNames) do
        local ok, module = pcall(require, name)
        if ok and module and type(module.new) == "function" then
            DecryptSkillDrive = module
            return DecryptSkillDrive
        end
    end

    if _G.DecryptSkillDrive and type(_G.DecryptSkillDrive.new) == "function" then
        DecryptSkillDrive = _G.DecryptSkillDrive
        return DecryptSkillDrive
    end

    return nil
end

local function getDriveInfoSafe(item)
    if not item then return nil end
    if GVDrive_Utils and GVDrive_Utils.getDriveInfo then
        local ok, info = pcall(GVDrive_Utils.getDriveInfo, item)
        if ok and info then return info end
    end
    return nil
end

local rarityFallback = {
    Facil = { en = "Easy", es = "Facil" },
    Moderado = { en = "Moderate", es = "Moderado" },
    Dificil = { en = "Hard", es = "Dificil" }
}

local function buildDriveDisplayName(isUSB, skillName, rarity)
    if not skillName then skillName = "Unknown" end
    if not rarity then rarity = "Unknown" end

    local baseKey = isUSB and "DisplayName_SkillDrive_" or "DisplayName_SkillFloppy_"
    local translationKey = baseKey .. skillName .. "_" .. rarity
    local label = getText and getText(translationKey) or translationKey
    if label == translationKey then
        local rarityLabel = rarityFallback[rarity]
        if rarityLabel then
            local language = "EN"
            if getCore and getCore() and getCore().getOptionLanguage then
                local ok, lang = pcall(function() return getCore():getOptionLanguage() end)
                if ok and lang then language = lang end
            end
            if language == "ES" then
                rarity = rarityLabel.es
            else
                rarity = rarityLabel.en
            end
        end
        label = string.format("%s (%s)", skillName, rarity)
    end
    return label
end

local function queueDecryptActions(player, worldObj, drives)
    if not drives or #drives == 0 then return end

    -- Validate laptop health before queuing actions
    local laptopItem = worldObj
    if laptopItem and laptopItem.getItem then
        local ok, inner = pcall(function() return laptopItem:getItem() end)
        if ok and inner then laptopItem = inner end
    end

    if LaptopSystem and laptopItem then
        local health = LaptopSystem.getLaptopHealth(laptopItem)
        if health <= 0 then
            player:Say(getTranslatedMessage("GVDrive_Msg_Laptop_Broken", "This laptop doesn't work... I should find a way to repair it."))
            return
        end
    end

    local playerObj = player
    if not playerObj or not playerObj.getInventory then return end

    local usbDrives = {}
    local floppyDrives = {}

    for _, drive in ipairs(drives) do
        if drive and drive.getItem and drive.getType then
            local item = drive:getItem()
            if item then
                local driveType = item:getType()
                if driveType == "USB" then
                    table.insert(usbDrives, drive)
                elseif driveType == "Floppy" then
                    table.insert(floppyDrives, drive)
                end
            end
        end
    end

    -- Sort drives by distance, closest first
    table.sort(usbDrives, function(a, b) return a:getDistanceTo(playerObj) < b:getDistanceTo(playerObj) end)
    table.sort(floppyDrives, function(a, b) return a:getDistanceTo(playerObj) < b:getDistanceTo(playerObj) end)

    -- Queue decrypt actions for USB drives first
    for _, drive in ipairs(usbDrives) do
        if drive and drive.getItem and drive.getType then
            local item = drive:getItem()
            if item then
                local driveSkill = item:getSkill()
                local driveRarity = item:getRarity()
                local action = ISUsbSys.queueDecryptAction(playerObj, drive)
                if action then
                    action:setActionAnim("Loot")
                    action:setOverrideHandModels(nil, nil)
                    action:setAnimVariable("LootPosition", "Belly")
                    action:setIgnoreIfFail(true)
                    action:setOnComplete(function()
                        -- Success or failure, remove one drive from the queue
                        if drive and drive.getItem and drive.getType then
                            local item = drive:getItem()
                            if item then
                                local driveType = item:getType()
                                if driveType == "USB" then
                                    -- For USB drives, we can directly remove the drive from the world
                                    -- This also ensures the drive is not used again in this session
                                    drive:removeFromWorld()
                                end
                            end
                        end
                    end)
                    playerObj:getQueue():add(action)
                end
            end
        end
    end

    -- Queue decrypt actions for Floppy drives
    for _, drive in ipairs(floppyDrives) do
        if drive and drive.getItem and drive.getType then
            local item = drive:getItem()
            if item then
                local driveSkill = item:getSkill()
                local driveRarity = item:getRarity()
                local action = ISFloppySys.queueDecryptAction(playerObj, drive)
                if action then
                    action:setActionAnim("Loot")
                    action:setOverrideHandModels(nil, nil)
                    action:setAnimVariable("LootPosition", "Belly")
                    action:setIgnoreIfFail(true)
                    action:setOnComplete(function()
                        -- Success or failure, remove one drive from the queue
                        if drive and drive.getItem and drive.getType then
                            local item = drive:getItem()
                            if item then
                                local driveType = item:getType()
                                if driveType == "Floppy" then
                                    -- For Floppy drives, we can directly remove the drive from the world
                                    -- This also ensures the drive is not used again in this session
                                    drive:removeFromWorld()
                                end
                            end
                        end
                    end)
                    playerObj:getQueue():add(action)
                end
            end
        end
    end
end

-- Function to start the decryption process
function startLaptopDecryption(player, laptop)
    if not player or not laptop then return end

    -- Validate laptop health
    local health = 100
    if LaptopSystem and laptop.getItem then
        local item = laptop:getItem()
        if item then
            health = LaptopSystem.getLaptopHealth(item)
        end
    end

    if health <= 0 then
        player:Say(getTranslatedMessage("GVDrive_Msg_Laptop_Broken", "This laptop doesn't work... I should find a way to repair it."))
        return
    end

    -- Check if already decrypting
    if player:getVariable("isDecrypting") == "true" then
        player:Say(getTranslatedMessage("GVDrive_Msg_AlreadyDecrypting", "I'm already decrypting a drive."))
        return
    end

    player:setVariable("isDecrypting", "true")

    -- Get all drives in the laptop
    local drives = {}
    if laptop.getItem and laptop.getItem().getInventory then
        local inv = laptop:getItem():getInventory()
        if inv then
            for i = 0, inv:getItems():size() - 1 do
                local item = inv:getItems():get(i)
                if item and (item:getType() == "USB" or item:getType() == "Floppy") then
                    table.insert(drives, item)
                end
            end
        end
    end

    -- If no drives found, show message
    if #drives == 0 then
        player:Say(getTranslatedMessage("GVDrive_Msg_NoDrives", "This laptop has no drives to decrypt."))
        player:setVariable("isDecrypting", "false")
        return
    end

    -- Queue decrypt actions for all drives
    queueDecryptActions(player, laptop, drives)

    -- Show final message with drive details
    local finalMessage = getTranslatedMessage("GVDrive_Msg_DecryptionStarted", "Decryption started for drives:")
    for _, drive in ipairs(drives) do
        if drive and drive.getItem then
            local item = drive:getItem()
            if item then
                local driveSkill = item:getSkill()
                local driveRarity = item:getRarity()
                local driveName = buildDriveDisplayName(drive:getType() == "USB", driveSkill, driveRarity)
                finalMessage = finalMessage .. "\n- " .. driveName
            end
        end
    end

    player:Say(finalMessage)
end

-- Function to stop the decryption process
function stopLaptopDecryption(player)
    if not player then return end

    player:setVariable("isDecrypting", "false")
    player:Say(getTranslatedMessage("GVDrive_Msg_DecryptionStopped", "Decryption process stopped."))
end

-- ISBaseTimedAction overrides
local function LaptopFillStart(action)
    -- Call the original start function
    action:originalStart()

    -- Additional custom logic
    -- For example, we can directly call the decryption without animation
    if action.character and action.character:getPlayerNum() >= 0 then
        local player = action.character
        local laptop = action.target

        -- Start the decryption process
        startLaptopDecryption(player, laptop)
    end
end

local function LaptopFillStop(action)
    -- Call the original stop function
    action:originalStop()

    -- Additional custom logic
    -- For example, we can stop the decryption if needed
    if action.character and action.character:getPlayerNum() >= 0 then
        local player = action.character

        -- Stop the decryption process
        stopLaptopDecryption(player)
    end
end

-- Patch the ISBaseTimedAction methods
local ISBaseTimedAction = ISBaseTimedAction
ISBaseTimedAction.originalStart = ISBaseTimedAction.start
ISBaseTimedAction.originalStop = ISBaseTimedAction.stop
ISBaseTimedAction.start = LaptopFillStart
ISBaseTimedAction.stop = LaptopFillStop
